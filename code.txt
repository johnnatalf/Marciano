// 1. Interface tradicional
interface AcaoPersonalizada {
    fun executar()
}

// Classe Base: Comandos Basicos
open class Marciano {

    open fun responda(frase: String): String {
        val fraseTratada = frase.trim()

        if (fraseTratada.isEmpty()) {
            return "Nao me incomode"
        }

        val contemLetras = fraseTratada.any { it.isLetter() }
        val ehGrito = contemLetras && fraseTratada == fraseTratada.toUpperCase()
        val ehPergunta = fraseTratada.endsWith("?")
        val contemEu = fraseTratada.toLowerCase().contains("eu")

        if (ehGrito && ehPergunta) {
            return "Relaxa, eu sei o que estou fazendo!"
        }
        
        if (ehGrito) {
            return "Opa! Calma ai!"
        }

        if (ehPergunta) {
            return "Certamente"
        }

        if (contemEu) {
            return "A responsabilidade e sua"
        }

        return "Tudo bem, como quiser"
    }
}

// 2. Classe de Extensao: Operacoes Matematicas Basicas
open class MarcianoMatematico : Marciano() {

    open fun responda(operacao: String, vararg operandos: Double): String {
        if (operandos.isEmpty()) return "Preciso de numeros para calcular."

        return when (operacao.toLowerCase()) {
            "some" -> "Essa eu sei, resultado: ${operandos.reduce { acc, d -> acc + d }}"
            "subtraia" -> "Essa eu sei, resultado: ${operandos.reduce { acc, d -> acc - d }}"
            "multiplique" -> "Essa eu sei, resultado: ${operandos.reduce { acc, d -> acc * d }}"
            "divida" -> {
                if (operandos.drop(1).any { it == 0.0 }) {
                    "Nao posso divisao por zero!"
                } else {
                    "Essa eu sei, resultado: ${operandos.reduce { acc, d -> acc / d }}"
                }
            }
            else -> "Nao entendi essa operacao matematica."
        }
    }
}

// 3. Classe de Extensao: Versao Premium
class MarcianoPremium(private val acaoCustomizada: AcaoPersonalizada) : MarcianoMatematico() {

    override fun responda(frase: String): String {
        val fraseTratada = frase.trim()
        
        if (fraseTratada.toLowerCase().contains("agir")) {
            acaoCustomizada.executar()
            return "E pra ja!"
        }
        
        return super.responda(frase)
    }
}

// 4. Programa Principal
fun main() {
    val acaoSecreta = object : AcaoPersonalizada {
        override fun executar() {
            println("\n[SISTEMA]: Iniciando protocolo de piadas espaciais...")
            println("Por que os alienigenas nao visitam a Terra? Porque leram as avaliacoes no TripAdvisor.")
        }
    }

    val robo = MarcianoPremium(acaoSecreta)

    println("==================================================")
    println("Bem-vindo ao Terminal do Robo Marciano Premium!")
    println("Voce pode conversar com ele, pedir contas ou faze-lo 'agir'.")
    println("Digite 'FIM' para encerrar a transmissao.")
    println("==================================================")

    while (true) {
        val entrada = readLine() ?: ""
        if (entrada.isEmpty()) break 

        println("\nVoce: $entrada")

        if (entrada.trim().toUpperCase() == "FIM") {
            println("\nMarciano: Tchau humano insignificante. Desconectando...")
            break
        }

        if (entrada.toLowerCase().startsWith("math ")) {
            val partes = entrada.split(" ")
            if (partes.size >= 3) {
                val operacao = partes[1]
                val numeros = partes.drop(2).mapNotNull { it.toDoubleOrNull() }.toDoubleArray()
                
                val resposta = robo.responda(operacao, *numeros)
                println("Marciano: $resposta")
            } else {
                println("Marciano: Formato matematico incorreto. Use: math [operacao] [numeros]")
            }
        } else {
            val resposta = robo.responda(entrada)
            println("Marciano: $resposta")
        }
    }
}
